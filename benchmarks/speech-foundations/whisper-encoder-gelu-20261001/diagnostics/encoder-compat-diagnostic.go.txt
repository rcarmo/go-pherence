// Test-only encoder plan rewriting: original tanh GELU and/or F16 storage
// rounding of encoder K/V immediately before attention.
func encCompatKernel(t *testing.T, env string) *vk.VkComputeKernel {
	path := os.Getenv(env)
	code, e := os.ReadFile(path)
	if e != nil {
		t.Fatal(e)
	}
	if fmt.Sprintf("%x", sha256.Sum256(code)) != os.Getenv(env+"_SHA") {
		t.Fatal("shader pin", env)
	}
	create := vk.VkKernelCreate
	if env == "ENCCOMPAT_ROUND_SPV" {
		create = vk.VkKernelCreateIntegerDot // admits GLSL.std.450 half pack/unpack
	}
	k, e := create(code, 2, 4)
	if e != nil {
		t.Fatal(e)
	}
	t.Cleanup(func() { k.Close() })
	return k
}

func encCompatFactory(t *testing.T, tanh, f16kv bool) func(context.Context, []vk.VkF32Stage) (*vk.VkF32Plan, error) {
	var gk, rk *vk.VkComputeKernel
	if tanh {
		gk = encCompatKernel(t, "ENCCOMPAT_GELU_SPV")
	}
	if f16kv {
		rk = encCompatKernel(t, "ENCCOMPAT_ROUND_SPV")
	}
	return func(ctx context.Context, s []vk.VkF32Stage) (*vk.VkF32Plan, error) {
		out := make([]vk.VkF32Stage, 0, len(s)+2)
		gelus, attns := 0, 0
		for _, st := range s {
			if len(st.Tensors) == 2 && st.Tensors[0] == st.Tensors[1] && len(st.PushWords) == 1 {
				gelus++
				if gk != nil {
					st.Kernel = gk
				}
			}
			if len(st.Tensors) == 4 && len(st.PushWords) == 5 && st.Tensors[0].Elements() == st.Tensors[1].Elements() && st.Tensors[1].Elements() == st.Tensors[2].Elements() {
				attns++
				if rk != nil {
					for _, kv := range st.Tensors[1:3] {
						n := kv.Elements()
						out = append(out, vk.VkF32Stage{Kernel: rk, Groups: [3]uint32{uint32((n + 255) / 256), 1, 1}, Tensors: []*vk.VkTensorF32{kv, kv}, PushWords: []uint32{uint32(n)}})
					}
				}
			}
			out = append(out, st)
		}
		ok := (len(s) == 5 && gelus == 2 && attns == 0) || (gelus == 1 && attns == 1) || (len(s) == 1 && gelus == 0 && attns == 0)
		if !ok {
			t.Fatalf("unexpected plan stages=%d gelus=%d attns=%d", len(s), gelus, attns)
		}
		return vk.NewVkF32Plan(ctx, out)
	}
}

func encCompatReadF32(t *testing.T, path string) []float32 {
	b, err := os.ReadFile(path)
	if err != nil || len(b)%4 != 0 {
		t.Fatal(path, err)
	}
	v := make([]float32, len(b)/4)
	for i := range v {
		v[i] = math.Float32frombits(binary.LittleEndian.Uint32(b[i*4:]))
	}
	return v
}

func TestWhisperEncoderCompatDiagnostic(t *testing.T) {
	prefixes := os.Getenv("ENCCOMPAT_PREFIXES") // prefix:lang,...
	if prefixes == "" {
		t.Skip("explicit diagnostic")
	}
	d, _ := t.Deadline()
	ctx, cancel := context.WithDeadline(context.Background(), d.Add(-time.Second))
	defer cancel()
	model, tok, policy, file := pinnedLegacyWhisperModelMode(t, ctx, true)
	defer file.Close()
	if !vk.VulkanInitIntegerDot() || !strings.Contains(vk.VulkanDeviceName(), os.Getenv("GO_PHERENCE_VULKAN_DEVICE")) {
		t.Fatal("device")
	}
	stats := func(a, b []float32) string {
		var mx, se float64
		for i := range a {
			d := float64(a[i] - b[i])
			se += d * d
			mx = math.Max(mx, math.Abs(d))
		}
		return fmt.Sprintf("max=%.4g rmse=%.6g", mx, math.Sqrt(se/float64(len(a))))
	}
	type arm struct {
		name        string
		tanh, f16kv bool
	}
	arms := []arm{{"base", false, false}, {"tanh", true, false}, {"f16kv", false, true}, {"tanh-f16kv", true, true}}
	type fixture struct {
		prefix, lang    string
		mel, def, nof16 []float32
	}
	var fx []fixture
	for _, p := range strings.Split(prefixes, ",") {
		parts := strings.SplitN(p, ":", 2)
		fx = append(fx, fixture{parts[0], parts[1], encCompatReadF32(t, parts[0]+"-orig-default-mel.f32"), encCompatReadF32(t, parts[0]+"-orig-default-hidden.f32"), encCompatReadF32(t, parts[0]+"-orig-nof16-hidden.f32")})
	}
	for _, a := range arms {
		enc, err := newVulkanEncoderPackedOnlyMode(ctx, model.Encoder, model.Config.MaxLength, encCompatFactory(t, a.tanh, a.f16kv), vulkanLinearOriginalQ5PaddedIntegerDotMMQ, file, true)
		if err != nil {
			t.Fatal(err)
		}
		for _, f := range fx {
			hidden, err := enc.Forward(ctx, f.mel)
			if err != nil {
				t.Fatal(err)
			}
			t.Logf("ENCCOMPAT %s arm=%s vs-default %s | vs-nof16 %s", f.prefix[strings.LastIndex(f.prefix, "/")+1:], a.name, stats(hidden, f.def), stats(hidden, f.nof16))
			buf := make([]byte, len(hidden)*4)
			for i, v := range hidden {
				binary.LittleEndian.PutUint32(buf[i*4:], math.Float32bits(v))
			}
			if err := os.WriteFile(f.prefix+"-enccompat-"+a.name+"-hidden.f32", buf, 0644); err != nil {
				t.Fatal(err)
			}
			for _, compat := range []bool{false, true} {
				v, e := checkedTimestampVocabulary(model.Config, tok, f.lang)
				if e != nil {
					t.Fatal(e)
				}
				opts, suppress, begin, e := resolvePCMGeneration(model.Config, v, PCMTranscribeOptions{Language: f.lang, Generation: policy}, model.Decoder.SuppressTokens, model.Decoder.BeginSuppressTokens)
				if e != nil {
					t.Fatal(e)
				}
				state, e := NewDecoderStateContext(ctx, model.Config, hidden, 1500, model.Decoder)
				if e != nil {
					t.Fatal(e)
				}
				if compat {
					state.applyOriginalDecoderCompatibility(1500)
				}
				segs, e := decodeCheckedTimestamps(ctx, model.Config, tok, v, opts, suppress, begin, func(token int) ([]float32, error) { return model.Decoder.ForwardToken(token, state), ctx.Err() })
				if e != nil {
					t.Fatal(e)
				}
				var b strings.Builder
				for _, s := range segs {
					fmt.Fprintf(&b, "[%.2f-%.2f %q]", s.Start, s.End, s.Text)
				}
				t.Logf("ENCCOMPAT_DECODE %s arm=%s compat=%v %s", f.prefix[strings.LastIndex(f.prefix, "/")+1:], a.name, compat, b.String())
			}
		}
		if err := enc.Close(); err != nil {
			t.Fatal(err)
		}
	}
}
