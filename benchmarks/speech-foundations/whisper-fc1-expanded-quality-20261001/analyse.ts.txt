function edits(ref:string[],hyp:string[]){
 const d=Array.from({length:ref.length+1},()=>new Array(hyp.length+1).fill(0));for(let i=0;i<=ref.length;i++)d[i][0]=i;for(let j=0;j<=hyp.length;j++)d[0][j]=j;
 for(let i=1;i<=ref.length;i++)for(let j=1;j<=hyp.length;j++)d[i][j]=Math.min(d[i-1][j]+1,d[i][j-1]+1,d[i-1][j-1]+(ref[i-1]===hyp[j-1]?0:1));
 let i=ref.length,j=hyp.length,S=0,D=0,I=0;const operations:any[]=[];while(i||j){if(i&&j&&ref[i-1]===hyp[j-1]&&d[i][j]===d[i-1][j-1]){i--;j--}else if(i&&j&&d[i][j]===d[i-1][j-1]+1){S++;operations.push({operation:'substitute',reference:ref[i-1],hypothesis:hyp[j-1]});i--;j--}else if(i&&d[i][j]===d[i-1][j]+1){D++;operations.push({operation:'delete',reference:ref[i-1]});i--}else{I++;operations.push({operation:'insert',hypothesis:hyp[j-1]});j--}}
 return {S,D,I,N:ref.length,total:d[ref.length][hyp.length],wer:ref.length?(S+D+I)/ref.length:null,operations:operations.reverse()};
}
