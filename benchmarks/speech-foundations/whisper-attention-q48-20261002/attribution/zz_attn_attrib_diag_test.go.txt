package vulkan

import (
	"context"
	"math"
	"math/rand"
	"os"
	"sort"
	"strconv"
	"strings"
	"testing"
	"time"
)

// Diagnostic only (archived after use): per-dispatch encoder attention time
// (16 dispatches per plan) and output bits versus the padded-extent kernel.
func TestZZAttentionAttributionDiagnostic(t *testing.T) {
	if os.Getenv("ATTNDIAG_VARIANTS") == "" {
		t.Skip("diagnostic")
	}
	d, _ := t.Deadline()
	ctx, cancel := context.WithDeadline(context.Background(), d.Add(-2*time.Second))
	defer cancel()
	if !VulkanInit() || !strings.Contains(VulkanDeviceName(), os.Getenv("GO_PHERENCE_VULKAN_DEVICE")) {
		t.Fatal("device")
	}
	const rows, heads, hd, reps = 1500, 20, 64, 16
	rng := rand.New(rand.NewSource(5))
	mk := func(scale float64) []float32 {
		v := make([]float32, rows*heads*hd)
		for i := range v {
			v[i] = float32(rng.NormFloat64() * scale)
		}
		return v
	}
	qv, kvv, vv := mk(1), mk(1), mk(1)
	a, err := NewVkTensorArena(ctx, 4*rows*heads*hd*5+(64<<10))
	if err != nil {
		t.Fatal(err)
	}
	defer a.Close()
	alloc := func(data []float32) *VkTensorF32 {
		x, e := a.AllocF32Aligned(ctx, 16, rows, heads*hd)
		if e != nil {
			t.Fatal(e)
		}
		if data != nil {
			if e := x.Upload(ctx, data); e != nil {
				t.Fatal(e)
			}
		}
		return x
	}
	tq, tk, tv, to := alloc(qv), alloc(kvv), alloc(vv), alloc(nil)
	var baseOut []float32
	for _, spec := range strings.Split(os.Getenv("ATTNDIAG_VARIANTS"), ",") {
		name, rest, _ := strings.Cut(spec, "=")
		path, tileText, _ := strings.Cut(rest, ":")
		tile, _ := strconv.Atoi(tileText)
		code := spirv_attention_f32_key32_padded_extent
		if path != "" {
			if code, err = os.ReadFile(path); err != nil {
				t.Fatal(err)
			}
		}
		op, err := newVkAttentionF32Code(ctx, code)
		if err != nil {
			t.Logf("ATTNDIAG %s create error: %v", name, err)
			continue
		}
		op.requiredHeadDim = 64
		st, err := op.Stage(ctx, to, tq, tk, tv, heads)
		if err != nil {
			t.Fatal(err)
		}
		st.Groups[0] = uint32((rows + tile - 1) / tile)
		var stages []VkF32Stage
		for i := 0; i < reps; i++ {
			stages = append(stages, st)
		}
		p, err := NewVkF32Plan(ctx, stages)
		if err != nil {
			t.Fatal(err)
		}
		var v []float64
		for r := 0; r < 6; r++ {
			start := time.Now()
			if err := p.Run(ctx); err != nil {
				t.Fatal(err)
			}
			if r > 0 {
				v = append(v, time.Since(start).Seconds()/reps*1000)
			}
		}
		sort.Float64s(v)
		out := make([]float32, rows*heads*hd)
		if err := to.Download(ctx, out); err != nil {
			t.Fatal(err)
		}
		cmp := "base"
		if baseOut == nil {
			baseOut = out
		} else {
			n := 0
			for i := range out {
				if math.Float32bits(out[i]) != math.Float32bits(baseOut[i]) {
					n++
				}
			}
			cmp = "BITEXACT"
			if n > 0 {
				cmp = "differ n=" + strconv.Itoa(n)
			}
		}
		t.Logf("ATTNDIAG %s tile=%d attn_ms=%.3f %s", name, tile, v[len(v)/2], cmp)
		if err := p.Close(); err != nil {
			t.Fatal(err)
		}
		if err := op.Close(); err != nil {
			t.Fatal(err)
		}
	}
}
