const rows = JSON.parse(await Bun.file(process.argv[2]).text());
const t: Record<string, number> = {};
for (const r of rows) { let k = r.op; if (r.op === "linear") k = "linear:" + r.inputs[1].split(".")[1]; if (r.op === "q8-activation") k = "q8:" + r.inputs[0]; t[k] = (t[k] ?? 0) + r.seconds / 3; }
let tot = 0; for (const [k, v] of Object.entries(t).sort((a, b) => b[1] - a[1])) { tot += v; console.log(k.padEnd(20), v.toFixed(3)); } console.log("total", tot.toFixed(3));
