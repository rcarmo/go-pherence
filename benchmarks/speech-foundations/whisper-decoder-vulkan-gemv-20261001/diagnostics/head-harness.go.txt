package whisper

import (
	"context"
	"errors"
	vk "github.com/rcarmo/go-pherence/backends/vulkan"
	"math"
	"os"
	"strings"
	"testing"
	"time"
)

func TestDecoderVulkanHeadDiagnostic(t *testing.T) {
	dir := os.Getenv("GO_PHERENCE_TEST_VK_DECODER_HEAD")
	if dir == "" {
		t.Skip("explicit pinned Vulkan head diagnostic")
	}
	deadline, ok := t.Deadline()
	if !ok || time.Until(deadline) > 120*time.Second {
		t.Fatal("120s")
	}
	ctx, cancel := context.WithDeadline(context.Background(), deadline.Add(-time.Second))
	defer cancel()
	t.Setenv("GO_PHERENCE_DISABLE_NVIDIA", "1")
	t.Setenv(envDecoderParallelRows, "1")
	model, _, _, file := pinnedLegacyWhisperModelMode(t, ctx, true)
	defer file.Close()
	k, n := model.Config.DecoderDModel, model.Config.VocabSize
	if k != 1280 || n != 51866 {
		t.Fatal("trainedshape")
	}
	if !vk.VulkanInit() || os.Getenv("GO_PHERENCE_VULKAN_DEVICE") == "" || !strings.Contains(vk.VulkanDeviceName(), os.Getenv("GO_PHERENCE_VULKAN_DEVICE")) {
		t.Fatal("physical")
	}
	before := vk.VulkanMemoryStats()
	defer func() {
		after := vk.VulkanMemoryStats()
		if after.Bytes != before.Bytes || after.Allocations != before.Allocations || after.InFlight || after.DeviceLost {
			t.Error("leak", before, after)
		}
	}()
	code, e := os.ReadFile(dir + "/gemv32.spv")
	if e != nil {
		t.Fatal(e)
	}
	kernel, e := vk.VkKernelCreate(code, 4, 8)
	if e != nil {
		t.Fatal(e)
	}
	defer func() {
		if e := kernel.Close(); e != nil {
			t.Error(e)
		}
	}()
	a, e := vk.NewVkTensorArena(ctx, (n*k+k+2*n+1024)*4)
	if e != nil {
		t.Fatal(e)
	}
	defer func() {
		if e := a.Close(); e != nil {
			t.Error(e)
		}
	}()
	alloc := func(shape ...int) *vk.VkTensorF32 {
		v, e := a.AllocF32(ctx, shape...)
		if e != nil {
			t.Fatal(e)
		}
		return v
	}
	left := alloc(256)
	guard := make([]float32, 256)
	for i := range guard {
		guard[i] = float32(i + 1)
	}
	if e := left.Upload(ctx, guard); e != nil {
		t.Fatal(e)
	}
	tx, tw, tb, out := alloc(k), alloc(n, k), alloc(n), alloc(n)
	right := alloc(256)
	if e := right.Upload(ctx, guard); e != nil {
		t.Fatal(e)
	}
	uploadStart := time.Now()
	if e := tw.Upload(ctx, model.Decoder.TokenEmbed); e != nil {
		t.Fatal(e)
	}
	t.Logf("DECODER_VK_HEAD_WEIGHT_UPLOAD seconds=%.9f bytes=%d", time.Since(uploadStart).Seconds(), len(model.Decoder.TokenEmbed)*4)
	plan, e := vk.NewVkF32Plan(ctx, []vk.VkF32Stage{{Kernel: kernel, Groups: [3]uint32{uint32((n + 31) / 32), 1, 1}, Tensors: []*vk.VkTensorF32{tx, tw, tb, out}, PushWords: []uint32{uint32(k), uint32(n)}}})
	if e != nil {
		t.Fatal(e)
	}
	defer func() {
		if e := plan.Close(); e != nil {
			t.Error(e)
		}
	}()
	rows := model.Config.MaxLength / 2
	encoded := make([]float32, rows*k)
	for i := range encoded {
		encoded[i] = float32(i%71-35) * .0027
	}
	state, e := NewDecoderStateContext(ctx, model.Config, encoded, rows, model.Decoder)
	if e != nil {
		t.Fatal(e)
	}
	check := func(a, b []float32) {
		t.Helper()
		if len(a) != len(b) {
			t.Fatal("extent")
		}
		for i, v := range a {
			if math.Float32bits(v) != math.Float32bits(b[i]) {
				t.Fatal("trainedlogitbits", i, v, b[i])
			}
		}
	}
	gpu := func(x []float32) []float32 {
		if e := tx.Upload(ctx, x); e != nil {
			t.Fatal(e)
		}
		if e := plan.Run(ctx); e != nil {
			t.Fatal(e)
		}
		got := make([]float32, n)
		if e := out.Download(ctx, got); e != nil {
			t.Fatal(e)
		}
		return got
	}
	for index, token := range []int{50258, 50259, 50359, 50364, 400, 370, 1029, 406} {
		ref := append([]float32(nil), model.Decoder.ForwardToken(token, state)...)
		x := append([]float32(nil), state.Bufs.normed...)
		check(ref, gpu(x))
		for repeat := 0; repeat < 5; repeat++ {
			for _, candidate := range []bool{repeat%2 == 1, repeat%2 == 0} {
				start := time.Now()
				var got []float32
				if candidate {
					got = gpu(x)
				} else {
					got = linearForwardOpt(x, model.Decoder.TokenEmbed, nil, 1, k, n)
				}
				wall := time.Since(start)
				check(ref, got)
				t.Logf("DECODER_VK_HEAD tokenIndex=%d candidate=%t repeat=%d seconds=%.9f values=%d", index, candidate, repeat, wall.Seconds(), n)
			}
		}
	}
	cancelled, stop := context.WithCancel(ctx)
	stop()
	if e := plan.Run(cancelled); !errors.Is(e, context.Canceled) {
		t.Fatal("precancel", e)
	}
	x := state.Bufs.normed
	check(linearForwardOpt(x, model.Decoder.TokenEmbed, nil, 1, k, n), gpu(x))
	for _, g := range []*vk.VkTensorF32{left, right} {
		got := make([]float32, 256)
		if e := g.Download(ctx, got); e != nil {
			t.Fatal(e)
		}
		check(guard, got)
	}
	t.Log("DECODER_VK_HEAD_ALL_LOGITS_EXACT8_PRE_CANCEL_REUSE_GUARDS_PASS")
}
