const [base,profile,input,name,out]=process.argv.slice(2);
const wav=await Bun.file(input).bytes();
const j=async(r:Response)=>{if(!r.ok)throw Error(r.status+' '+await r.text());return r.json()};
const t0=performance.now();
const job=await j(await fetch(`${base}/v1/jobs?`+new URLSearchParams({profile,name}),{method:'POST',headers:{'Content-Type':'application/octet-stream'},body:wav}));
await j(await fetch(`${base}/v1/jobs/${job.id}/enqueue`,{method:'POST'}));
let s:any;for(let i=0;i<600;i++){s=await j(await fetch(`${base}/v1/jobs/${job.id}`));if(!['pending','queued','running'].includes(s.status))break;await Bun.sleep(250)}
const secs=(performance.now()-t0)/1000;
const crypto=await import('node:crypto');let ok=0;
await Bun.$`mkdir -p ${out}`;
await Bun.write(`${out}/complete.json`,JSON.stringify(s,null,1));
for(const a of s.artifacts??[]){const r=await fetch(`${base}/v1/jobs/${job.id}/artifacts/${a.name}`);const b=new Uint8Array(await r.arrayBuffer());const h=crypto.createHash('sha256').update(b).digest('hex');await Bun.write(`${out}/${a.name}`,b);if(b.length===a.bytes&&h===a.sha256&&r.headers.get('etag')===`"sha256-${a.sha256}"`)ok++}
console.log(JSON.stringify({profile,name,id:job.id,status:s.status,wall:+secs.toFixed(2),artifacts:s.artifacts?.length,verified:ok,error:s.error}));
