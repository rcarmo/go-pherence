package vulkan

import (
	"encoding/binary"
	"os"
	"strings"
	"testing"
)

func TestZZMMQVariantAdmission(t *testing.T) {
	for _, p := range strings.Split(os.Getenv("MMQDIAG_SPV"), ",") {
		if p == "" {
			continue
		}
		b, err := os.ReadFile(p)
		if err != nil {
			t.Fatal(err)
		}
		w := make([]uint32, len(b)/4)
		for i := range w {
			w[i] = binary.LittleEndian.Uint32(b[4*i:])
		}
		c, err := vkInspectSPIRVMode(w, true)
		t.Logf("ADMIT %s err=%v local=%v dot=%v", p[strings.LastIndex(p, "/")+1:], err, c.LocalSize, c.IntegerDot)
	}
}
