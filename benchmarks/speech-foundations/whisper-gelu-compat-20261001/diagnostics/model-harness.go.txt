package whisper

import (
	"context"
	"crypto/sha256"
	"fmt"
	vk "github.com/rcarmo/go-pherence/backends/vulkan"
	"os"
	"testing"
)

// Test-only substitution restricted to the explicit padded-key benchmark mode.
func geluCompatibilityFactory(t *testing.T, ctx context.Context) (func(context.Context, []vk.VkF32Stage) (*vk.VkF32Plan, error), string) {
	path := os.Getenv("GO_PHERENCE_TEST_ENCODER_GELU_SPV")
	if path == "" {
		return vk.NewVkF32Plan, ""
	}
	code, e := os.ReadFile(path)
	if e != nil {
		t.Fatal(e)
	}
	pin := fmt.Sprintf("%x", sha256.Sum256(code))
	if pin != os.Getenv("GO_PHERENCE_TEST_ENCODER_GELU_SHA") {
		t.Fatal("diagnosticshaderpin")
	}
	k, e := vk.VkKernelCreate(code, 2, 4)
	if e != nil {
		t.Fatal(e)
	}
	t.Cleanup(func() {
		if e := k.Close(); e != nil {
			t.Error(e)
		}
	})
	factory := func(ctx context.Context, s []vk.VkF32Stage) (*vk.VkF32Plan, error) {
		indices := []int{}
		switch len(s) {
		case 5:
			indices = []int{1, 3}
		case 12:
			indices = []int{9}
		case 1:
		default:
			t.Fatal("unexpectedencoderplan")
		}
		for _, i := range indices {
			if len(s[i].Tensors) != 2 || len(s[i].PushWords) != 1 {
				t.Fatal("GELUstageABI")
			}
			s[i].Kernel = k
		}
		return vk.NewVkF32Plan(ctx, s)
	}
	return factory, pin
}
